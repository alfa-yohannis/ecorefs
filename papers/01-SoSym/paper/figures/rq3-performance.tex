% Figure: mean save and load times for generated models (RQ3).
% Data: eval/results/benchmark_summary.md, 30 measured runs per model size on an
% otherwise idle machine, with the node in offline mode and an empty datastore.
% Hue separates the two persistence paths (categorical slots 1 and 2 of the
% validated default palette, checked with validate_palette.js against a white
% page). Line style and marker separate save from load, so the figure stays
% readable without color. The same values appear in Table tab:rq3.
\documentclass[border=2pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\definecolor{ink}{HTML}{0B0B0B}
\definecolor{inksoft}{HTML}{52514E}
\definecolor{axisgray}{HTML}{C3C2B7}
\definecolor{gridgray}{HTML}{E1E0D9}
\definecolor{seriesone}{HTML}{2A78D6}
\definecolor{seriestwo}{HTML}{EB6834}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=7.8cm, height=5.8cm,
  font=\footnotesize,
  xmode=log, ymode=log,
  log basis x=10, log basis y=10,
  xmin=60, xmax=900000,
  ymin=0.4, ymax=2000,
  xtick={100,1000,10000,100000,500000},
  xticklabels={100,1K,10K,100K,500K},
  ytick={1,10,100,1000},
  yticklabels={1,10,100,{1,000}},
  xlabel={Model size (elements)}, ylabel={Mean time (ms)},
  axis line style={draw=axisgray}, tick style={draw=axisgray},
  axis x line*=bottom, axis y line*=left,
  ymajorgrids, major grid style={draw=gridgray, line width=0.4pt},
  label style={text=inksoft}, tick label style={text=inksoft},
  legend style={draw=none, fill=none, legend columns=2, at={(0.5,1.04)}, anchor=south,
    /tikz/every even column/.append style={column sep=8pt}},
  legend cell align=left,
]
\addplot[seriesone, line width=0.9pt, mark=*, mark size=2pt]
  coordinates {(100,0.69) (1000,0.99) (10000,3.10) (100000,31.50) (500000,311.14)};
\addplot[seriestwo, line width=0.9pt, mark=square*, mark size=2pt]
  coordinates {(100,21.20) (1000,21.28) (10000,25.07) (100000,77.73) (500000,423.25)};
\addplot[seriesone, densely dashed, line width=0.9pt, mark=triangle*, mark size=2.4pt]
  coordinates {(100,2.16) (1000,2.51) (10000,7.81) (100000,69.41) (500000,505.03)};
\addplot[seriestwo, densely dashed, line width=0.9pt, mark=diamond*, mark size=2.4pt]
  coordinates {(100,2.43) (1000,3.31) (10000,10.23) (100000,88.86) (500000,683.87)};
\legend{XMI save, IPFS save, XMI load, IPFS load}
\end{axis}
\end{tikzpicture}
\end{document}
